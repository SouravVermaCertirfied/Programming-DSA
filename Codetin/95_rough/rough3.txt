#include<stdio.h>

pthread_mutex_t lock;
pthread_cond_t cond;

int fuel = 0;

void *car(void *args){

    while(1){
        pthread_mutex_lock(&lock);
        if(fuel>45){
            fuel -= 45;
            break;
        }
        pthread_cond_wait(&cond);
    }
    pthread_mutex_unlock(&lock);
    return NULL;
}

void *tanker(void *args){
    while(1){
        pthread_mutex_lock(&lock);
        if(fuel>100)
            break;
        fuel += 5;
        pthread_cond_signal(&cond, &lock);
    }
    pthread_mutex_unlock(&lock);
    return NULL;
}

int main(){
    pthread_t c, t;

    pthread_mutex_init(&lock, NULL);
    pthread_cond_init(&cond, NULL);
    
    pthread_create(&c, NULL, car, NULL);
    pthread_create(&t, NULL, tanker, NULL);

    pthread_join(c, NULL);
    pthread_join(t, NULL);

    return 0;
}
