#include<stdio.h>
#include<pthread.h>

int curr = 0;
pthread_mutex_t lock;

void *tanker(void *args){
    
    pthread_mutex_lock(&lock);
    while(curr<100){
        curr++;
        if(curr>=100) 
            break;
    }
    pthread_mutex_lock(&lock);
    return NULL;
}

void *car(void *args){

    while(1){
        pthread_mutex_lock(&lock);
        if(curr>45=){
            curr-= 45;
            break;
        }
        pthread_mutex_unlock(&lock);
    }
    pthread_mutex_unlock(&lock);
    return NULL;
}

int main(){
    pthread mycar, mytanker;
    pthread_mutex_init(&lock);

    pthread_create(&mycar, NULL, car, NULL);
    pthread_create(&mytanker, NULL, tanker, NULL);

    pthread_join(mycar, NULL);
    pthread_join(mytanker, NULL);

    pthread_mutex_destroy(&lock);
    return 0;
}
