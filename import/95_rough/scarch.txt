#include<pthread.h>
#include<stdio.h>

int fuel = 0;
pthread_mutex_t lock;
pthread_cond_t cond;

void *tanker(void *args){
    pthread_mutex_lock(&lock);
    while(fuel < 100){
        fuel += 10;
        pthread_cond_signal(&cond);
    }
    pthread_mutex_unlock(&lock);
    return NULL;
}

void *car(void *args){
    pthread_mutex_lock(&lock)
    while(1){
        if(fuel>=45){
            fuel -= 45;
            break;
        }
        pthread_cond_wait(&cond, &lock);
    }
    pthread_mutex_unlock(&lock);
    return NULL;
}

int main(){
    pthread_t t , c;

    pthread_mutex_init(&lock, NULL);
    pthread_cond_init(&cond, NULL);

    pthread_create(&t, NULL, tanker, NULL);
    pthread_create(&c, NULL, car, NULL);

    pthread_join(t, NULL);
    pthread_join(c, NULL);

    pthread_mutex_destroy(&lock);
    pthread_cond_destroy(&cond);

    return 0;
}
